"""
peft_methods.py

LoRA configuration and utilities for Medical VQA.

Final experimental method: LoRA only.
"""

from peft import LoraConfig, get_peft_model


def create_lora_config(config):
    """Create LoRA configuration from project YAML settings."""

    peft_config = config["peft"]

    if peft_config.get("method") != "lora":
        raise ValueError("This project uses LoRA only.")

    lora = peft_config.get("lora", {})

    if not lora.get("enabled", True):
        raise ValueError("LoRA is disabled in the configuration.")

    return LoraConfig(
        r=int(lora.get("r", 8)),
        lora_alpha=int(lora.get("alpha", 16)),
        lora_dropout=float(lora.get("dropout", 0.05)),
        target_modules=lora.get(
            "target_modules", ["q_proj", "v_proj"]
        ),
        bias=lora.get("bias", "none"),
        task_type=lora.get("task_type", "CAUSAL_LM"),
    )


def apply_lora(model, config):
    """Apply LoRA to a base model."""

    lora_config = create_lora_config(config)

    model = get_peft_model(model, lora_config)

    # Keep trainable adapter parameters in FP32.
    for name, parameter in model.named_parameters():
        if parameter.requires_grad and "lora_" in name:
            parameter.data = parameter.data.float()

    model.print_trainable_parameters()

    return model


def get_trainable_parameter_summary(model):
    """Return total and trainable parameter counts."""

    total = sum(
        parameter.numel()
        for parameter in model.parameters()
    )

    trainable = sum(
        parameter.numel()
        for parameter in model.parameters()
        if parameter.requires_grad
    )

    return {
        "total_parameters": total,
        "trainable_parameters": trainable,
        "trainable_percentage": (
            (trainable / total) * 100 if total else 0
        ),
    }

"""
train_pipeline.py

Prepares Medical VQA annotations for LLaVA-style
multimodal fine-tuning.

Input:
    data/processed/annotations/*.jsonl

Output:
    data/processed/llava_format/*.json
"""

import argparse
import json
from pathlib import Path

import yaml


PROJECT_ROOT = Path(__file__).resolve().parent
CONFIG_PATH = PROJECT_ROOT / "configs" / "config.yaml"

DATASET_PREFIXES = {
    "SLAKE": "slake",
    "VQA_RAD": "vqa_rad",
    "PathVQA": "pathvqa",
}

SPLITS = ("train", "validation", "test")


def load_config():
    """Load and validate the project configuration."""

    if not CONFIG_PATH.is_file():
        raise FileNotFoundError(
            f"Configuration not found: {CONFIG_PATH}"
        )

    with CONFIG_PATH.open("r", encoding="utf-8") as file:
        config = yaml.safe_load(file)

    if not isinstance(config, dict):
        raise ValueError("Invalid YAML configuration.")

    for section in ("project", "paths", "datasets"):
        if section not in config:
            raise ValueError(
                f"Missing configuration section: {section}"
            )

    return config


def resolve_project_path(path_value):
    """Resolve a configured path relative to the project root."""

    path = Path(path_value)

    if path.is_absolute():
        raise ValueError(
            f"Configured paths must be project-relative: {path}"
        )

    resolved = (PROJECT_ROOT / path).resolve()

    try:
        resolved.relative_to(PROJECT_ROOT.resolve())
    except ValueError as error:
        raise ValueError(
            f"Path escapes project directory: {path}"
        ) from error

    return resolved


def get_annotation_path(dataset, split):
    """Return the JSONL annotation file for a dataset split."""

    prefix = DATASET_PREFIXES[dataset]

    return (
        PROJECT_ROOT
        / "data"
        / "processed"
        / "annotations"
        / f"{prefix}_{split}.jsonl"
    )


def load_records(annotation_path, expected_dataset, expected_split):
    """Read and validate JSONL annotation records."""

    if not annotation_path.is_file():
        raise FileNotFoundError(
            f"Annotation file not found: {annotation_path}"
        )

    records = []

    required_fields = {
        "dataset",
        "split",
        "image_path",
        "question",
        "answer",
    }

    with annotation_path.open(
        "r",
        encoding="utf-8-sig",
    ) as file:

        for line_number, line in enumerate(file, start=1):

            if not line.strip():
                continue

            try:
                record = json.loads(line)

            except json.JSONDecodeError as error:
                raise ValueError(
                    f"Invalid JSON at line {line_number} "
                    f"in {annotation_path.name}"
                ) from error

            if not isinstance(record, dict):
                raise ValueError(
                    f"Expected JSON object at line {line_number}"
                )

            missing = required_fields - record.keys()

            if missing:
                raise ValueError(
                    f"Missing fields at line {line_number}: "
                    f"{sorted(missing)}"
                )

            if record["dataset"] != expected_dataset:
                raise ValueError(
                    f"Dataset mismatch at line {line_number}"
                )

            if record["split"] != expected_split:
                raise ValueError(
                    f"Split mismatch at line {line_number}"
                )

            if not str(record["question"]).strip():
                raise ValueError(
                    f"Empty question at line {line_number}"
                )

            if not str(record["answer"]).strip():
                raise ValueError(
                    f"Empty answer at line {line_number}"
                )

            records.append(record)

    return records


def convert_record(record, expected_dataset, expected_split):
    """
    Convert one annotation into the LLaVA conversation format.
    """

    relative_image_path = Path(record["image_path"])

    if relative_image_path.is_absolute():
        raise ValueError(
            f"Absolute image path is not allowed: "
            f"{relative_image_path}"
        )

    image_path = resolve_project_path(
        relative_image_path
    )

    if not image_path.is_file():
        raise FileNotFoundError(
            f"Image does not exist: {image_path}"
        )

    return {
        "id": str(
            record.get(
                "source_id",
                f"{expected_dataset}_{expected_split}",
            )
        ),
        "image": relative_image_path.as_posix(),
        "conversations": [
            {
                "from": "human",
                "value": (
                    "<image>\n"
                    + str(record["question"]).strip()
                ),
            },
            {
                "from": "gpt",
                "value": str(record["answer"]).strip(),
            },
        ],
        "metadata": {
            key: value
            for key, value in record.items()
            if key not in {
                "image_path",
                "question",
                "answer",
                "dataset",
                "split",
                "source_id",
            }
        },
    }


def prepare_split(
    dataset,
    split,
    output_dir,
    limit=None,
):
    """Convert and save one dataset split."""

    annotation_path = get_annotation_path(
        dataset,
        split,
    )

    records = load_records(
        annotation_path,
        dataset,
        split,
    )

    if limit is not None:
        if limit < 1:
            raise ValueError("--limit must be positive.")

        records = records[:limit]

    converted_records = [
        convert_record(record, dataset, split)
        for record in records
    ]

    output_dir.mkdir(
        parents=True,
        exist_ok=True,
    )

    output_path = (
        output_dir
        / f"{DATASET_PREFIXES[dataset]}_{split}.json"
    )

    with output_path.open(
        "w",
        encoding="utf-8",
    ) as file:

        json.dump(
            converted_records,
            file,
            indent=2,
            ensure_ascii=False,
        )

    print(
        f"{dataset:10} | {split:10} | "
        f"{len(converted_records):6} records | "
        f"Saved: {output_path.relative_to(PROJECT_ROOT)}"
    )

    return len(converted_records)


def main():

    parser = argparse.ArgumentParser(
        description="Prepare Medical VQA data for LLaVA."
    )

    parser.add_argument(
        "--dataset",
        choices=["all", "SLAKE", "VQA_RAD", "PathVQA"],
        default="all",
    )

    parser.add_argument(
        "--split",
        choices=["all", "train", "validation", "test"],
        default="all",
    )

    parser.add_argument(
        "--limit",
        type=int,
        default=None,
    )

    args = parser.parse_args()

    config = load_config()

    # Use the explicit llava_format path.
    output_dir = resolve_project_path(
        config["paths"].get(
            "llava_format",
            "data/processed/llava_format",
        )
    )

    datasets = (
        list(DATASET_PREFIXES)
        if args.dataset == "all"
        else [args.dataset]
    )

    splits = (
        list(SPLITS)
        if args.split == "all"
        else [args.split]
    )

    total_records = 0

    print("\nMedical VQA training-data preparation")
    print("-------------------------------------")

    for dataset in datasets:

        dataset_config = config["datasets"].get(dataset)

        if dataset_config is None:
            raise ValueError(
                f"{dataset} is missing from config.yaml"
            )

        if not dataset_config.get("enabled", False):
            print(f"Skipping disabled dataset: {dataset}")
            continue

        for split in splits:

            count = prepare_split(
                dataset=dataset,
                split=split,
                output_dir=output_dir,
                limit=args.limit,
            )

            total_records += count

    print("-------------------------------------")
    print(f"Total records prepared: {total_records}")
    print(f"Output directory: {output_dir}")
    print("\nData preparation completed successfully.")
    print("No model training was performed.")


if __name__ == "__main__":
    main()

from pathlib import Path

import torch

PROJECT_ROOT = Path(__file__).resolve().parent

def load_model_config(config_path=CONFIG_PATH):
    """Load and validate the project configuration."""

    config_path = Path(config_path)

    if not config_path.is_absolute():
        config_path = PROJECT_ROOT / config_path

    if not config_path.is_file():
        raise FileNotFoundError(
            f"Configuration not found: {config_path}"
        )

    with config_path.open("r", encoding="utf-8") as file:
        config = yaml.safe_load(file)

    required = [
        "project",
        "model",
        "training",
        "peft",
        "datasets",
        "paths",
    ]

    for section in required:
        if section not in config:
            raise ValueError(
                f"Missing configuration section: {section}"
            )

    model_config = config["model"]

    if model_config.get("model_family") != "llava":
        raise ValueError("Only LLaVA model family is supported.")

    if config["peft"].get("method") != "lora":
        raise ValueError("Final project supports LoRA only.")

    if not model_config.get("model_id"):
        raise ValueError("Missing model.model_id.")

    return config


def get_torch_dtype(dtype_name):
    """Convert a dtype string to a PyTorch dtype."""

    supported = {
        "float16": torch.float16,
        "float32": torch.float32,
        "bfloat16": torch.bfloat16,
    }

    if dtype_name not in supported:
        raise ValueError(
            f"Unsupported torch dtype: {dtype_name}"
        )

    if dtype_name == "bfloat16" and not torch.cuda.is_available():
        raise RuntimeError(
            "BFloat16 configuration requires a compatible GPU."
        )

    return supported[dtype_name]


def get_device():
    """Detect the available computing device."""

    return "cuda" if torch.cuda.is_available() else "cpu"


def load_llava_model(
    config=None,
    adapter_path=None,
    device_map=None,
):
    """
    Load the LLaVA-Med base model and optionally a LoRA adapter.

    Returns:
        tokenizer, model, image_processor, context_length
    """

    if config is None:
        config = load_model_config()

    model_config = config["model"]
    model_id = model_config["model_id"]

    if device_map is None:
        device_map = (
            model_config.get("device_map", "auto")
            if torch.cuda.is_available()
            else "cpu"
        )

    try:
        from llava.model.builder import load_pretrained_model
        from llava.mm_utils import get_model_name_from_path

    except ImportError as error:
        raise ImportError(
            "Official LLaVA code is unavailable. "
            "Install the LLaVA repository and its dependencies "
            "before loading the model."
        ) from error

    model_name = get_model_name_from_path(model_id)

    print("Loading LLaVA-Med base model...")
    print("Model:", model_id)
    print("Device:", device_map)

    tokenizer, model, image_processor, context_length = (
        load_pretrained_model(
            model_path=model_id,
            model_base=None,
            model_name=model_name,
            device_map=device_map,
        )
    )

    if adapter_path is not None:
        adapter_path = Path(adapter_path)

        if not adapter_path.is_dir():
            raise FileNotFoundError(
                f"LoRA adapter directory not found: {adapter_path}"
            )

        if not (adapter_path / "adapter_config.json").is_file():
            raise FileNotFoundError(
                f"adapter_config.json missing in {adapter_path}"
            )

        try:
            from peft import PeftModel

        except ImportError as error:
            raise ImportError(
                "PEFT is required to load the LoRA adapter."
            ) from error

        print("Loading LoRA adapter:", adapter_path)

        model = PeftModel.from_pretrained(
            model,
            str(adapter_path),
            is_trainable=False,
        )

    model.eval()

    return tokenizer, model, image_processor, context_length


def count_parameters(model):
    """Count total and trainable model parameters."""

    total = sum(
        parameter.numel()
        for parameter in model.parameters()
    )

    trainable = sum(
        parameter.numel()
        for parameter in model.parameters()
        if parameter.requires_grad
    )

    return {
        "total_parameters": total,
        "trainable_parameters": trainable,
        "trainable_percentage": (
            100 * trainable / total if total else 0
        ),
    }

"""
dataset.py

PyTorch Dataset and DataLoader implementation
for the Medical VQA research project.

Supported datasets:
    - SLAKE
    - VQA_RAD
    - PathVQA
"""

import json
from pathlib import Path
from typing import Callable, Optional

from PIL import Image, UnidentifiedImageError
from torch.utils.data import Dataset, DataLoader

from utils import PROJECT_ROOT


# --------------------------------------------------
# 1. DATASET CONFIGURATION
# --------------------------------------------------

DATASET_PREFIXES = {
    "SLAKE": {
        "prefix": "slake",
        "label": "SLAKE",
    },
    "VQA_RAD": {
        "prefix": "vqa_rad",
        "label": "VQA_RAD",
    },
    "PATHVQA": {
        "prefix": "pathvqa",
        "label": "PathVQA",
    },
}

VALID_SPLITS = ("train", "validation", "test")


# --------------------------------------------------
# 2. MEDICAL VQA DATASET
# --------------------------------------------------

class MedicalVQADataset(Dataset):

    def __init__(
        self,
        dataset_name: str,
        split: str,
        transform: Optional[Callable] = None,
        max_samples: Optional[int] = None,
    ):

        dataset_key = dataset_name.strip().upper()
        self.split = split.strip().lower()
        self.transform = transform

        # Validate dataset name
        if dataset_key not in DATASET_PREFIXES:
            raise ValueError(
                f"Unsupported dataset: {dataset_name}. "
                f"Choose from SLAKE, VQA_RAD, PathVQA."
            )

        self.dataset_name = DATASET_PREFIXES[dataset_key]["label"]
        prefix = DATASET_PREFIXES[dataset_key]["prefix"]

        # Validate split
        if self.split not in VALID_SPLITS:
            raise ValueError(
                f"Invalid split: {self.split}. "
                f"Choose from {VALID_SPLITS}."
            )

        # Locate annotation file
        self.annotation_file = (
            PROJECT_ROOT
            / "data"
            / "processed"
            / "annotations"
            / f"{prefix}_{self.split}.jsonl"
        )

        if not self.annotation_file.is_file():
            raise FileNotFoundError(
                f"Annotation file not found: {self.annotation_file}"
            )

        # Load annotations
        self.records = self._load_annotations()

        if max_samples is not None:
            if max_samples < 1:
                raise ValueError("max_samples must be positive.")

            self.records = self.records[:max_samples]

        if not self.records:
            raise ValueError(
                f"No records found for "
                f"{self.dataset_name} ({self.split})."
            )

        print(
            f"Loaded {len(self.records)} records: "
            f"{self.dataset_name} / {self.split}"
        )

    # --------------------------------------------------
    # 3. LOAD JSONL ANNOTATIONS
    # --------------------------------------------------

    def _load_annotations(self):

        records = []

        with self.annotation_file.open(
            "r",
            encoding="utf-8-sig",
        ) as file:

            for line_number, line in enumerate(file, start=1):

                if not line.strip():
                    continue

                try:
                    record = json.loads(line)

                except json.JSONDecodeError as error:
                    raise ValueError(
                        f"Invalid JSON in "
                        f"{self.annotation_file.name}, "
                        f"line {line_number}: {error}"
                    ) from error

                if not isinstance(record, dict):
                    raise ValueError(
                        f"Record at line {line_number} "
                        f"must be a JSON object."
                    )

                required_fields = {
                    "dataset",
                    "split",
                    "image_path",
                    "question",
                    "answer",
                }

                missing = required_fields - record.keys()

                if missing:
                    raise ValueError(
                        f"Missing fields at line {line_number}: "
                        f"{sorted(missing)}"
                    )

                if record["dataset"] != self.dataset_name:
                    raise ValueError(
                        f"Dataset mismatch at line {line_number}: "
                        f"expected {self.dataset_name}, "
                        f"got {record['dataset']}."
                    )

                if record["split"] != self.split:
                    raise ValueError(
                        f"Split mismatch at line {line_number}: "
                        f"expected {self.split}, "
                        f"got {record['split']}."
                    )

                if not str(record["question"]).strip():
                    raise ValueError(
                        f"Empty question at line {line_number}."
                    )

                if not str(record["answer"]).strip():
                    raise ValueError(
                        f"Empty answer at line {line_number}."
                    )

                records.append(record)

        return records

    # --------------------------------------------------
    # 4. DATASET LENGTH
    # --------------------------------------------------

    def __len__(self):
        return len(self.records)

    # --------------------------------------------------
    # 5. LOAD INDIVIDUAL SAMPLE
    # --------------------------------------------------

    def __getitem__(self, index):

        record = self.records[index]

        relative_image_path = Path(record["image_path"])

        if relative_image_path.is_absolute():
            raise ValueError(
                "Absolute image paths are not allowed."
            )

        image_path = (
            PROJECT_ROOT / relative_image_path
        ).resolve()

        # Prevent paths from escaping the project directory
        try:
            image_path.relative_to(PROJECT_ROOT.resolve())

        except ValueError as error:
            raise ValueError(
                f"Image path escapes project directory: "
                f"{record['image_path']}"
            ) from error

        if not image_path.is_file():
            raise FileNotFoundError(
                f"Medical image not found: {image_path}"
            )

        # Open and convert image to RGB
        try:
            with Image.open(image_path) as image:
                image = image.convert("RGB")

        except (
            UnidentifiedImageError,
            OSError,
            ValueError,
        ) as error:

            raise RuntimeError(
                f"Unable to load image: {image_path}"
            ) from error

        # Apply optional image transformation
        if self.transform is not None:
            image = self.transform(image)

        return {
            "image": image,
            "question": str(record["question"]),
            "answer": str(record["answer"]),
            "dataset": self.dataset_name,
            "split": self.split,
            "question_type": record.get("question_type"),
            "answer_type": record.get("answer_type"),
            "modality": record.get("modality"),
            "source_id": record.get("source_id"),
            "source_image_id": record.get("source_image_id"),
        }


# --------------------------------------------------
# 6. CUSTOM BATCH COLLATION

def medical_vqa_collate_fn(batch):

    return {
        "images": [item["image"] for item in batch],
        "questions": [item["question"] for item in batch],
        "answers": [item["answer"] for item in batch],
        "datasets": [item["dataset"] for item in batch],
        "splits": [item["split"] for item in batch],
        "question_types": [
            item["question_type"] for item in batch
        ],
        "answer_types": [
            item["answer_type"] for item in batch
        ],
        "modalities": [
            item["modality"] for item in batch
        ],
        "source_ids": [
            item["source_id"] for item in batch
        ],
        "source_image_ids": [
            item["source_image_id"] for item in batch
        ],
    }


# --------------------------------------------------
# 7. DATALOADER FACTORY
# --------------------------------------------------

def build_dataloader(
    dataset_name: str,
    split: str,
    batch_size: int = 1,
    shuffle: Optional[bool] = None,
    num_workers: int = 0,
    transform: Optional[Callable] = None,
    max_samples: Optional[int] = None,
):

    if batch_size < 1:
        raise ValueError("batch_size must be at least 1.")

    if num_workers < 0:
        raise ValueError("num_workers cannot be negative.")

    if shuffle is None:
        shuffle = split.strip().lower() == "train"

    dataset = MedicalVQADataset(
        dataset_name=dataset_name,
        split=split,
        transform=transform,
        max_samples=max_samples,
    )

    return DataLoader(
        dataset,
        batch_size=batch_size,
        shuffle=shuffle,
        num_workers=num_workers,
        collate_fn=medical_vqa_collate_fn,
        pin_memory=False,
    )
